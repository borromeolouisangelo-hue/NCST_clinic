    /**
     * All regions in official order.
     *
     * @return array<string>
     */
    public static function regions(): array
    {
        return array_keys(self::regionsWithProvinces());
    }

    /**
     * Get provinces for a given region.
     *
     * @param  string  $region
     * @return array<string>
     */
    public static function provinces(?string $region = null): array
{
    $data = self::regionsWithProvinces();

    if ($region !== null && $region !== '') {
        return $data[$region] ?? [];
    }

    return self::allProvinces();
}

    /**
     * All provinces in alphabetical order.
     *
     * @return array<string>
     */
    public static function allProvinces(): array
    {
        $provinces = [];
        foreach (self::regionsWithProvinces() as $provinces) {
            $provinces = array_merge($provinces, $provinces);
        }
        sort($provinces);
        return array_unique($provinces);
    }

    /**
     * Get region for a given province.
     *
     * @param  string  $province
     * @return string|null
     */
    public static function regionOfProvince(string $province): ?string
    {
        foreach (self::regionsWithProvinces() as $region => $provinces) {
            if (in_array($province, $provinces, true)) {
                return $region;
            }
        }
        return null;
    }

    /**
     * Check if a province belongs to a region.
     *
     * @param  string  $region
     * @param  string  $province
     * @return bool
     */
    public static function provinceInRegion(string $region, string $province): bool
    {
        return in_array($province, self::provinces($region), true);
    }

    /**
     * Check if a province is valid.
     *
     * @param  string  $province
     * @return bool
     */
    public static function isValidProvince(string $province): bool
    {
        return isset(self::regionsWithProvinces()[$province]);
    }
